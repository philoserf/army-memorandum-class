% Coverage example: \enclsnocount. See #43 and #52.
% Shipped in 0.3.0 with zero in-repo uses. It suppresses the enclosure count per
% AR 25-50 Figure 4-4, so "Encl" prints without the leading number.
%
% #43 proposes collapsing the three \am@encls branches; this is the baseline that
% would demonstrate such a change is safe.
\documentclass{../armymemo}

\address{Organizational Name/Title}
\address{Standardized Street Address}
\address{CITY, STATE~~12345-1234}

\author{John W. Smith}\rank{CPT}\branch{CY}
\officesymbol{ABC-DEF-GH}
\signaturedate{10 April 2019}
\memoline{MEMORANDUM FOR RECORD}
\documentmark{UNCLASSIFIED//FOR OFFICIAL USE ONLY (EXAMPLE ONLY)}
\subject{A memorandum with an uncounted enclosure list}
\title{COMMANDING}
\enclsnocount
\addencl{AR 25-50}
\addencl{DA Form 1234}

\begin{document}

\begin{enumerate}
\item Two enclosures are declared, but \verb|\enclsnocount| suppresses the leading
  count, so the block heading reads Encls rather than 2 Encls. The entries
  themselves stay numbered.
\item A second paragraph, so the body is a well-formed AR 25-50 list.
\end{enumerate}

\end{document}
