% Coverage example: exactly one \address. See #26 and #52.
% All other examples supply three address lines; this is the single-line path.
%
% #26 is fixed. This file's golden used to record the bug deliberately -- supplying
% one address appended three placeholder lines to it, producing a four-line
% letterhead that began with the real address and continued with boilerplate. The
% `make golden` diff that removed those three lines is the proof the fix worked,
% which is exactly why the baseline was captured wrong on purpose.
%
% The golden now shows one address line, which is what the document asks for.
\documentclass{../armymemo}

\address{Organizational Name/Title}

\author{John W. Smith}\rank{CPT}\branch{CY}
\officesymbol{ABC-DEF-GH}
\signaturedate{10 April 2019}
\memoline{MEMORANDUM FOR RECORD}
\documentmark{UNCLASSIFIED//FOR OFFICIAL USE ONLY (EXAMPLE ONLY)}
\subject{A memorandum with a single address line}
\title{COMMANDING}
\addencl{AR 25-50}

\begin{document}

\begin{enumerate}
\item This memorandum supplies exactly one address line.
\item A second paragraph, so the body is a well-formed AR 25-50 list.
\end{enumerate}

\end{document}
